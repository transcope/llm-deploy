# =============================================================================
# LLM-Deploy 依赖清单
# =============================================================================
# 用途：从零安装时参考。以下版本均已在本项目 V100 容器 zetta_ld 中跑通验证。
#
# 注意：
#   1. V100 不支持 bfloat16，安装 torch 时选 CUDA 12.4 版本即可
#   2. 项目采用双虚拟环境隔离：
#      - venv-quant（量化环境）：含 gptqmodel 2.0.0 等量化工具链
#      - venv-deploy（部署评测环境）：无 gptqmodel，避免 optimum 版本冲突
#      详见 docs/V100_SERVER_GUIDE.md 第 3 节
#   3. gptqmodel==2.0.0 是 CUDA 12.4 torch 2.6 定制版（whl 安装，容器内编译）。
#      宿主机 CUDA 12.6，但容器运行时仅依赖容器内 CUDA 12.4（向下兼容）。
#      此版本与 optimum（要求 gptqmodel>=7.0.0）不兼容。
#      部署评测环境（venv-deploy）不含 gptqmodel，无此冲突。
#   4. llmcompressor/compressed-tensors 方案需要 SM>=80（A100+），V100 不可用。
#      V100 生产链路：gptqmodel 后端 + 标准 GPTQ 格式 + vLLM Exllama kernel。
#   5. 评测脚本仅依赖 torch + transformers + accelerate，不需要 lm-eval。
#   6. 容器内 torch==2.6.0+cu124，宿主机 CUDA 12.6（容器 CUDA 可低于宿主机）。
#
# 环境重建：
#   量化环境：     pip install -r requirements-quant.txt
#   部署评测环境： pip install -r requirements-deploy.txt
# =============================================================================

# ===== 基础框架 =====
# 容器实测: torch==2.6.0+cu124（CUDA 12.4, 与 xformers 0.0.28.post3 兼容）
torch>=2.5.1,<2.7.0             # PyTorch（V100 CUDA 12.4 验证通过）
torchvision>=0.20.1             # PyTorch Vision
torchaudio>=2.5.1               # PyTorch Audio
transformers>=4.51.0            # HuggingFace Transformers（>=4.51 才支持 Qwen3）
accelerate>=0.34.0              # 模型加载加速（评测时建议 device_map=None 避免钩子）
xformers==0.0.28.post3          # V100 上替代 FlashAttention-2

# ===== 推理引擎 =====
# 容器实测: vllm==0.8.3（V100 使用 V0 引擎，V1 需要 SM>=8.0）
# V100+Qwen3 无原生 vLLM 实现，回退 Transformers 后端，性能低于原生模型。
vllm>=0.8.0                     # vLLM 推理引擎

# ===== 量化工具链 =====
# V100 生产推荐：GPTQ 方案（gptqmodel 后端 → 标准 GPTQ 格式 → vLLM Exllama kernel）
# V100 不可用：llmcompressor（SM>=80）、AWQ GEMM（SM>=75）
# 详见 docs/CALIBRATION_GUIDE.md 和 install_quant_tools.sh
#
# 【关键版本锁定】gptqmodel
# 容器实测: gptqmodel==2.0.0+cu124torch2.5（定制 whl，非 PyPI 版本）
# 此版本适配 CUDA 12.4 + PyTorch 2.6，MODEL_MAP 不含 qwen3，
# 需搭配 qwen3_gptq_adapter.py 使用（quantize_model.py 自动注入）。
# 注意：与 optimum 不兼容（optimum 要求 gptqmodel>=7.0.0）。
gptqmodel==2.0.0                # ⚠️ 锁定 2.0.0，升级前需验证 Qwen3 兼容性

llmcompressor==0.4.0            # A100+ 量化工具（V100 不可用，仅用于 compressed-tensors 格式输出）
compressed-tensors==0.9.0       # SmoothQuant W8A8 部署格式支持（A100+）
autoawq<0.2.7                   # AWQ 量化 (legacy)
bitsandbytes>=0.49.0            # 8-bit/4-bit 动态量化（NF4 路径，V100 兼容）

# ===== 其他依赖 =====
lm-eval>=0.4.5                  # EleutherAI lm-evaluation-harness（可选，标准 Benchmark）
openai>=1.30.0                  # OpenAI API 客户端（调用 vLLM 服务）
Pillow>=10.0.0                  # 图像处理
requests>=2.32.0                # HTTP 请求
pyyaml>=6.0                     # YAML 配置文件解析
datasets>=3.0.0                 # HuggingFace Datasets（校准数据）
numpy>=1.26.0,<2.0.0            # vllm 0.8.x 尚未支持 numpy 2.x
scipy>=1.14.0
tqdm>=4.66.0

# ===== 可选: 开发调试 =====
ipykernel>=6.29.0               # Jupyter 支持
matplotlib>=3.9.0               # 可视化
pandas>=2.2.0                   # 数据分析
